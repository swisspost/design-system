{/* Full-width page section with an h2 title, optional content and an optional call-to-action link. */}
{/* Props: id (used for the heading and aria-labelledby), title, palette (default | alternate | brand | accent), */}
{/* icon (optional post-icon name), iconPosition (inline | top, defaults to inline; top centers the icon above the title), */}
{/* titleAlign (start | center, defaults to start), className (optional extra classes for the section), */}
{/* href, linkText, linkIcon (defaults to arrowright). */}

<section
  className={[
    'docs-band palette',
    `palette-${props.palette ?? 'default'}`,
    props.children || props.href ? 'py-48' : 'py-32',
    props.className,
  ]
    .filter(Boolean)
    .join(' ')}
  aria-labelledby={props.id}
>
  <div className="container">
    {props.icon && props.iconPosition === 'top' && (
      <div className="d-flex justify-content-center mb-16">
        <post-icon
          name={props.icon}
          className="w-48 h-48 w-md-64 h-md-64"
          aria-hidden="true"
        ></post-icon>
      </div>
    )}
    <h2
      id={props.id}
      className={[
        'mt-0',
        !props.children && !props.href && 'mb-0',
        props.titleAlign === 'center' && 'text-center',
        props.icon && props.iconPosition !== 'top' && 'd-flex align-items-center gap-16',
      ]
        .filter(Boolean)
        .join(' ')}
    >
      {props.icon && props.iconPosition !== 'top' && (
        <post-icon
          name={props.icon}
          className="w-48 h-48 flex-shrink-0"
          aria-hidden="true"
        ></post-icon>
      )}
      {props.title}
    </h2>
    {props.children}
    {props.href && (
      <a className="btn btn-primary w-full w-sm-auto" href={props.href}>
        {props.linkText}
        <post-icon name={props.linkIcon ?? 'arrowright'} aria-hidden="true"></post-icon>
      </a>
    )}
  </div>
</section>
