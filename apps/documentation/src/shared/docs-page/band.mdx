      className={[
        'mt-0',

      ]
        .filter(Boolean)
        .join(' ')}
    >
      {props.icon && (
        <post-icon
          name={props.icon}
          className="w-48 h-48 flex-shrink-0"
          aria-hidden="true"
        ></post-icon>
      )}

      <a className="btn btn-primary w-full w-sm-auto" href={props.href}>

      </a>
    )}
